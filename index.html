// Splash screen : affiché une seule fois par session window.addEventListener('load', function () { const splash = document.getElementById('splash'); if (!splash) return; // Le splash a déjà été affiché ? if (sessionStorage.getItem('splashShown')) { // On le masque immédiatement splash.style.display = 'none'; return; } // Première visite : on mémorise sessionStorage.setItem('splashShown', 'true'); // On laisse le splash 3 secondes setTimeout(function () { splash.classList.add('hidden'); // On le retire complètement du DOM setTimeout(function () { splash.style.display = 'none'; }, 800); }, 3000); });
